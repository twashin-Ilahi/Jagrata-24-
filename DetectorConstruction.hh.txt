/** 
 * @brief Advanced DetectorConstruction (Grandmaster Level)
 *
 * This class constructs the complete detector geometry for the Relativistic Muon Lifetime Measurements Experiment.
 * It provides full dynamic 3D configuration for all detector components, optimized material caching (including composite
 * materials for PMT and SiPM), and includes the DarkBox with internal detectors, along with advanced features such as
 * overlap checking, visualization, GDML loading (placeholder), and a single trigger volume.
 */
class G4VPhysicalVolume;
class DetectorConstruction : public G4VUserDetectorConstruction {
public:
    DetectorConstruction();
    virtual ~DetectorConstruction();

    /// Constructs and returns the world volume.
    virtual G4VPhysicalVolume* Construct();

    //=== PRIMARY FUNCTIONS ===
    void ResetSensitiveDetectors();
    void UpdateGeometryParameters();
    void AdvancedOverlapCheck();
    void DumpAdvancedGeometryInfo(const G4String& fileName);
    void SetAdvancedVisAttributes();
    void ToggleVisualization();

    //=== ADVANCED ENHANCEMENTS ===
    void AdvancedDynamicGeometryUpdate();
    void AdvancedGeometryEnhancement();
    void IntegrateAllSensitiveDetectors();
    void LoadGeometryFromGDML(const G4String& gdmlFile);
    void OptimizeMaterials();
    void BuildTriggerVolume();

    /// Central builder integrating all features.
    G4VPhysicalVolume* BuildAdvancedDetector();

private:
    /// Define and initialize all materials (with caching and composite definitions).
    void DefineMaterials();

    // Runtime configuration messenger.
    G4GenericMessenger* fMessenger;

    // Global material cache.
    std::map<G4String, G4Material*> fMatCache;

    // Dynamic geometry parameters (in meters).
    G4double fWorldSizeZ;    ///< World half-length in Z.
    // Scintillator1.
    G4double fScin1PosX;     ///< X position for Scintillator1.
    G4double fScin1PosY;     ///< Y position for Scintillator1.
    G4double fScin1PosZ;     ///< Z position for Scintillator1.
    // MRPC1.
    G4double fMRPC1PosX;     ///< X position for MRPC1.
    G4double fMRPC1PosY;     ///< Y position for MRPC1.
    G4double fMRPC1PosZ;     ///< Z position for MRPC1.
    // MRPC2.
    G4double fMRPC2PosX;     ///< X position for MRPC2.
    G4double fMRPC2PosY;     ///< Y position for MRPC2.
    G4double fMRPC2PosZ;     ///< Z position for MRPC2.
    // Scintillator6 (Halo).
    G4double fScin6PosX;     ///< X position for Scintillator6.
    G4double fScin6PosY;     ///< Y position for Scintillator6.
    G4double fScin6PosZ;     ///< Z position for Scintillator6.
    // Scintillator5.
    G4double fScin5PosX;     ///< X position for Scintillator5.
    G4double fScin5PosY;     ///< Y position for Scintillator5.
    G4double fScin5PosZ;     ///< Z position for Scintillator5.

    G4double fDarkBoxPosX;   ///< X position for DarkBox.
    G4double fDarkBoxPosY;   ///< Y position for DarkBox.
    G4double fDarkBoxPosZ;   ///< Z position for DarkBox.
    // Internal DarkBox volumes.
    G4double fScin2PosX;     ///< X position for internal Scintillator2.
    G4double fScin2PosY;     ///< Y position for internal Scintillator2.
    G4double fScin2PosZ;     ///< Z position for internal Scintillator2.
    G4double fScin3PosX;     ///< X position for internal Scintillator3.
    G4double fScin3PosY;     ///< Y position for internal Scintillator3.
    G4double fScin3PosZ;     ///< Z position for internal Scintillator3.
    G4double fScin4PosX;     ///< X position for internal Scintillator4.
    G4double fScin4PosY;     ///< Y position for internal Scintillator4.

//---------------------------------------------------------------------
// DefineMaterials: Populate fMatCache with basic and composite materials.
void DetectorConstruction::DefineMaterials() {
    G4NistManager* nist = G4NistManager::GetNistManager();
    fMatCache.clear();

    fMatCache["G4_AIR"] = nist->FindOrBuildMaterial("G4_AIR");
    fMatCache["G4_PLASTIC_SC_VINYLTOLUENE"] = nist->FindOrBuildMaterial("G4_PLASTIC_SC_VINYLTOLUENE");
    fMatCache["G4_Al"] = nist->FindOrBuildMaterial("G4_Al");

    // Composite materials for PMT and SiPM.
    G4double densityPMTGlass = 2.2*g/cm3;
    fMatCache["PMTGlass"] = new G4Material("PMTGlass", densityPMTGlass, 2);
    fMatCache["PMTGlass"]->AddMaterial(nist->FindOrBuildMaterial("G4_SILICON_DIOXIDE"), 0.9);
    fMatCache["PMTGlass"]->AddMaterial(nist->FindOrBuildMaterial("G4_SODIUM"), 0.1);

    G4double densitySiPM = 2.33*g/cm3;
    fMatCache["SiPMComposite"] = new G4Material("SiPMComposite", densitySiPM, 2);
    fMatCache["SiPMComposite"]->AddMaterial(nist->FindOrBuildMaterial("G4_Si"), 0.95);
    fMatCache["SiPMComposite"]->AddMaterial(nist->FindOrBuildMaterial("G4_PLASTIC_SC_VINYLTOLUENE"), 0.05);

    // Fallback basic materials.
    fMatCache["G4_SILICON_DIOXIDE"] = nist->FindOrBuildMaterial("G4_SILICON_DIOXIDE");
    fMatCache["G4_Si"] = nist->FindOrBuildMaterial("G4_Si");

    // DWCGas.
    G4double densityDWCGas = 1.2e-3*g/cm3;
    fMatCache["DWCGas"] = new G4Material("DWCGas", densityDWCGas, 2, kStateGas, 293.15*kelvin, 1.0*atmosphere);
    fMatCache["DWCGas"]->AddMaterial(nist->FindOrBuildMaterial("G4_Ar"), 0.80);
    fMatCache["DWCGas"]->AddMaterial(nist->FindOrBuildMaterial("G4_CARBON_DIOXIDE"), 0.20);

    // MRPCGas.
    G4Element* elC = nist->FindOrBuildElement("C");
    G4Element* elH = nist->FindOrBuildElement("H");
    G4Element* elF = nist->FindOrBuildElement("F");
    G4Element* elS = nist->FindOrBuildElement("S");
    G4double densityR134a = 4.25e-3*g/cm3;
    G4Material* R134a = new G4Material("R134a", densityR134a, 3, kStateGas, 293.15*kelvin, 1.0*atmosphere);
    R134a->AddElement(elC, 2);
    R134a->AddElement(elH, 2);
    R134a->AddElement(elF, 4);
    G4double densityIso = 2.5e-3*g/cm3;
    G4Material* Isobutane = new G4Material("Isobutane", densityIso, 2, kStateGas, 293.15*kelvin, 1.0*atmosphere);
    Isobutane->AddElement(elC, 4);
    Isobutane->AddElement(elH, 10);
    G4double densitySF6 = 6.17e-3*g/cm3;
    G4Material* SF6 = new G4Material("SF6", densitySF6, 2, kStateGas, 293.15*kelvin, 1.0*atmosphere);
    SF6->AddElement(elS, 1);
    SF6->AddElement(elF, 6);
    G4double densityMRPCGas = 1.2e-3*g/cm3;
    G4Material* MRPCGas = new G4Material("MRPCGas", densityMRPCGas, 3, kStateGas, 293.15*kelvin, 1.0*atmosphere);
    MRPCGas->AddMaterial(R134a, 0.90);
    MRPCGas->AddMaterial(Isobutane, 0.05);
    MRPCGas->AddMaterial(SF6, 0.05);
    fMatCache["MRPCGas"] = MRPCGas;
}

//---------------------------------------------------------------------
// BuildAdvancedDetector: Construct the complete detector geometry.
G4VPhysicalVolume* DetectorConstruction::BuildAdvancedDetector() {
    G4double zOffset = fZOffset;
    G4NistManager* nist = G4NistManager::GetNistManager();

    // --- World Volume ---
    G4double world_x = 2*m, world_y = 2*m, world_z = fWorldSizeZ;
    G4Box* solidWorld = new G4Box("World", world_x, world_y, world_z);
    G4LogicalVolume* logicWorld = new G4LogicalVolume(solidWorld, fMatCache["G4_AIR"], "World");
    G4VPhysicalVolume* physWorld = new G4PVPlacement(0, G4ThreeVector(0,0,0), logicWorld, "World", 0, false, 0, true);

    // --- Scintillator1 (20x10x0.5 cm³) ---
    G4Box* solidScin1 = new G4Box("Scintillator1", 20*cm/2, 10*cm/2, 0.5*cm/2);
    G4LogicalVolume* logicScin1 = new G4LogicalVolume(solidScin1, fMatCache["G4_PLASTIC_SC_VINYLTOLUENE"], "Scintillator1");
    G4ThreeVector posScin1(fScin1PosX, fScin1PosY, fScin1PosZ - zOffset);
    new G4PVPlacement(0, posScin1, logicScin1, "Scintillator1", logicWorld, false, 0, true);

    // --- DWC1 (10x10x5 cm³) ---
    G4Box* solidDWC1 = new G4Box("DWC1", 10*cm/2, 10*cm/2, 5*cm/2);
    G4LogicalVolume* logicDWC1 = new G4LogicalVolume(solidDWC1, fMatCache["DWCGas"], "DWC1");
    G4ThreeVector posDWC1(fDWC1PosX, fDWC1PosY, fScin1PosZ - zOffset + fDWC1PosZ);
    new G4PVPlacement(0, posDWC1, logicDWC1, "DWC1", logicWorld, false, 0, true);

    // --- Scintillator6 (Halo, 20x10x0.5 cm³ with 4x4 cm² hole) ---
    G4Box* solidScin6 = new G4Box("Scintillator6", 20*cm/2, 10*cm/2, 0.5*cm/2);
    G4Box* solidHole = new G4Box("Hole", 4*cm/2, 4*cm/2, 0.6*cm/2);
    G4SubtractionSolid* solidScin6WithHole = new G4SubtractionSolid("Scintillator6WithHole", solidScin6, solidHole, 0, G4ThreeVector(0,0,0));
    G4LogicalVolume* logicScin6 = new G4LogicalVolume(solidScin6WithHole, fMatCache["G4_PLASTIC_SC_VINYLTOLUENE"], "Scintillator6");
    G4ThreeVector posScin6(fScin6PosX, fScin6PosY, fScin6PosZ - zOffset);
    new G4PVPlacement(0, posScin6, logicScin6, "Scintillator6", logicWorld, false, 0, true);

    // --- MRPC1 (30x30x0.5 cm³) ---
    G4Box* solidMRPC1 = new G4Box("MRPC1", 30*cm/2, 30*cm/2, 0.5*cm/2);
    G4LogicalVolume* logicMRPC1 = new G4LogicalVolume(solidMRPC1, fMatCache["MRPCGas"], "MRPC1");
    G4ThreeVector posMRPC1(fMRPC1PosX, fMRPC1PosY, fMRPC1PosZ - zOffset);
    new G4PVPlacement(0, posMRPC1, logicMRPC1, "MRPC1", logicWorld, false, 0, true);

    // --- DarkBox (Full dimensions: 100x50x100 cm³; half-dimensions: 50, 25, 50 cm) ---
    G4Box* solidDarkBox = new G4Box("DarkBox", 50*cm, 25*cm, 50*cm);
    G4LogicalVolume* logicDarkBox = new G4LogicalVolume(solidDarkBox, fMatCache["G4_Al"], "DarkBox");
    G4ThreeVector posDarkBox(fDarkBoxPosX, fDarkBoxPosY, fDarkBoxPosZ - zOffset);
    new G4PVPlacement(0, posDarkBox, logicDarkBox, "DarkBox", logicWorld, false, 0, true);

    // --- Internal DarkBox Volumes ---
    // Scintillator2 (20x10x0.5 cm³)

//---------------------------------------------------------------------
// IntegrateAllSensitiveDetectors: Attach sensitive detectors to key volumes.
void DetectorConstruction::IntegrateAllSensitiveDetectors() {
    const std::vector<G4String> volNames = {
        "Scintillator1", "MRPC1", "DWC1", "Scintillator6", "MRPC2", "Scintillator5",
        "Scintillator2", "Scintillator3", "Scintillator4", "Trigger",
        "PMT2", "PMT3", "PMT4", "SiPM2", "SiPM3", "SiPM4"
    };
    G4SDManager* sdManager = G4SDManager::GetSDMpointer();
    for (const auto& name : volNames) {
        G4LogicalVolume* lv = G4LogicalVolumeStore::GetInstance()->GetVolume(name);
        if (lv && !lv->GetSensitiveDetector()) {
            MyAdvancedSensitiveDetector* sd = new MyAdvancedSensitiveDetector("SD_" + name);
            sdManager->AddNewDetector(sd);
            lv->SetSensitiveDetector(sd);
        }
    }
}

//---------------------------------------------------------------------
// AdvancedOverlapCheck: Check for overlaps using configurable parameters.
void DetectorConstruction::AdvancedOverlapCheck() {
    G4GeometryManager::GetInstance()->OpenGeometry();
    auto pvStore = G4PhysicalVolumeStore::GetInstance();
    for (auto pv : *pvStore) {
        if (pv->GetLogicalVolume()->CheckOverlaps(fOverlapMaxChecks, 0, true, fOverlapTolerance)) {
            G4Exception("DetectorConstruction::AdvancedOverlapCheck", "OverlapError",
                        FatalException, ("Overlap detected in volume: " + pv->GetName()).c_str());
        }
    }
}

//---------------------------------------------------------------------
// DumpAdvancedGeometryInfo: Dump detailed geometry info to a file.
void DetectorConstruction::DumpAdvancedGeometryInfo(const G4String& fileName) {
    std::ofstream outFile(fileName);
    if (!outFile.is_open()) {
        G4Exception("DetectorConstruction::DumpAdvancedGeometryInfo", "FileError",
                    JustWarning, "Unable to open file for advanced geometry dump.");
        return;
    }
    outFile << "=== Advanced Geometry Dump ===\n";
    auto pvStore = G4PhysicalVolumeStore::GetInstance();
    for (auto pv : *pvStore) {
        outFile << "Volume: " << pv->GetName()
                << ", Material: " << pv->GetLogicalVolume()->GetMaterial()->GetName() << "\n";
    }
    outFile.close();
    G4UImanager::GetUIpointer()->ApplyCommand("/geometry/dumpStructure");
    G4cout << "Advanced geometry info dumped to " << fileName << G4endl;
}

// SetAdvancedVisAttributes: Apply visualization attributes to key volumes.
void DetectorConstruction::SetAdvancedVisAttributes() {
    // For Scintillator1.
    G4LogicalVolume* scin1LV = G4LogicalVolumeStore::GetInstance()->GetVolume("Scintillator1");
    if (scin1LV) {
        G4VisAttributes* visScin1 = new G4VisAttributes(G4Colour(0.0, 1.0, 0.0)); // Green
        visScin1->SetForceSolid(true);
        visScin1->SetVisibility(true);
        visScin1->SetLineWidth(2);
        scin1LV->SetVisAttributes(visScin1);
    }
    // For MRPC1.
    G4LogicalVolume* mrpc1LV = G4LogicalVolumeStore::GetInstance()->GetVolume("MRPC1");
    if (mrpc1LV) {
        G4VisAttributes* visMRPC1 = new G4VisAttributes(G4Colour(0.0, 0.0, 1.0)); // Blue
        visMRPC1->SetForceSolid(true);
        visMRPC1->SetVisibility(true);
        visMRPC1->SetLineWidth(2);
        mrpc1LV->SetVisAttributes(visMRPC1);
    }
    // Default visualization for volumes without attributes.
    auto lvStore = G4LogicalVolumeStore::GetInstance();
    for (auto lv : *lvStore) {
        if (!lv->GetVisAttributes()) {
            G4VisAttributes* defaultVis = new G4VisAttributes(G4Colour(0.8, 0.8, 0.8));
            defaultVis->SetForceSolid(true);
            lv->SetVisAttributes(defaultVis);
        }
    }
}

//---------------------------------------------------------------------
// AdvancedGeometryEnhancement: Verify geometry integrity and boundaries.
void DetectorConstruction::AdvancedGeometryEnhancement() {
    G4LogicalVolume* worldLV = G4LogicalVolumeStore::GetInstance()->GetVolume("World");
    if (!worldLV) {
        G4Exception("DetectorConstruction::AdvancedGeometryEnhancement", "WorldVolumeError",
                    FatalException, "World logical volume not found!");
    }
    auto pvStore = G4PhysicalVolumeStore::GetInstance();
    for (auto pv : *pvStore) {
        G4ThreeVector pos = pv->GetTranslation();
        if (std::abs(pos.x()) > 2*m || std::abs(pos.y()) > 2*m || std::abs(pos.z()) > fWorldSizeZ) {
            G4Exception("DetectorConstruction::AdvancedGeometryEnhancement", "BoundaryError",
                        FatalException, ("Volume " + pv->GetName() + " lies outside world boundaries.").c_str());
        }
    }
    G4cout << "Advanced geometry enhancement check passed: All volumes within boundaries." << G4endl;
}
